cmake_minimum_required(VERSION 3.22)
project(LibertyOmniscience VERSION 0.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
option(OMNI_PLUGIN "Build JUCE plugin" ON)
find_package(Threads REQUIRED)
enable_testing()
add_executable(CosmosTests tests/CosmosTests.cpp)
target_include_directories(CosmosTests PRIVATE Source)
target_link_libraries(CosmosTests PRIVATE Threads::Threads)
add_test(NAME CosmosFoundation COMMAND CosmosTests)
add_executable(CosmosBenchmark tests/CosmosBenchmark.cpp)
target_include_directories(CosmosBenchmark PRIVATE Source)
target_link_libraries(CosmosBenchmark PRIVATE Threads::Threads)
add_executable(CosmosFilterQA tests/CosmosFilterQA.cpp)
target_include_directories(CosmosFilterQA PRIVATE Source)
target_link_libraries(CosmosFilterQA PRIVATE Threads::Threads)
if(MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8 AND (CMAKE_GENERATOR_PLATFORM STREQUAL "x64" OR CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64)$"))
  add_library(CosmosAVX2 STATIC Source/CosmosAVX2.cpp)
  target_compile_options(CosmosAVX2 PRIVATE /arch:AVX2)
  foreach(t CosmosTests CosmosBenchmark CosmosFilterQA)
    target_link_libraries(${t} PRIVATE CosmosAVX2)
    target_compile_definitions(${t} PRIVATE COSMOS_AVX2_DISPATCH=1)
  endforeach()
endif()
if(OMNI_PLUGIN)
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
  set(formats VST3 Standalone)
  if(APPLE)
    list(APPEND formats AU)
  endif()
  juce_add_plugin(Omniscience
    COMPANY_NAME "Liberty Music Software"
    BUNDLE_ID com.libertymusicsoftware.omniscience
    PLUGIN_MANUFACTURER_CODE Lbty
    PLUGIN_CODE Omns
    PRODUCT_NAME "Omniscience"
    IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE
    FORMATS ${formats})
  target_sources(Omniscience PRIVATE Source/Plugin.cpp)
  if(TARGET CosmosAVX2)
    target_link_libraries(Omniscience PRIVATE CosmosAVX2)
    target_compile_definitions(Omniscience PRIVATE COSMOS_AVX2_DISPATCH=1)
  endif()
  add_executable(CosmosPresetQA tests/CosmosPresetQA.cpp Source/Plugin.cpp)
  target_compile_definitions(CosmosPresetQA PRIVATE COSMOS_PRESET_QA=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(CosmosPresetQA PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
  if(TARGET CosmosAVX2)
    target_link_libraries(CosmosPresetQA PRIVATE CosmosAVX2)
    target_compile_definitions(CosmosPresetQA PRIVATE COSMOS_AVX2_DISPATCH=1)
  endif()
  target_compile_definitions(Omniscience PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(Omniscience PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
endif()
